CREATE TABLE public.tafsirs (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  surah smallint NOT NULL CHECK (surah >= 1 AND surah <= 114),
  ayah smallint NOT NULL CHECK (ayah > 0),
  round_number smallint NOT NULL DEFAULT 1,
  content text NOT NULL CHECK (char_length(content) > 0),
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  updated_at timestamp with time zone NOT NULL DEFAULT now(),
  tags ARRAY NOT NULL DEFAULT '{}'::text[],
  user_id uuid DEFAULT auth.uid(),
  CONSTRAINT tafsirs_pkey PRIMARY KEY (id)
);
CREATE TABLE public.comments (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  tafsir_id uuid NOT NULL,
  guest_name text NOT NULL DEFAULT 'ناشناس'::text CHECK (char_length(guest_name) <= 60),
  content text NOT NULL CHECK (char_length(content) > 0 AND char_length(content) <= 2000),
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  user_id uuid DEFAULT auth.uid(),
  CONSTRAINT comments_pkey PRIMARY KEY (id),
  CONSTRAINT comments_tafsir_id_fkey FOREIGN KEY (tafsir_id) REFERENCES public.tafsirs(id)
);
